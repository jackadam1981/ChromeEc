cmake_minimum_required(VERSION 3.13.1)
project(math)

find_package(Cros REQUIRED PATHS ${CMAKE_CURRENT_SOURCE_DIR}/../../)

# Copy math_util.h so we don't add the full include directory to the include
# path
configure_file(
		${PLATFORM_EC}/include/math_util.h ${CMAKE_BINARY_DIR}/include/math_util.h
)

# Add the common test files and the math_util.c
list(APPEND SOURCES
		${PLATFORM_EC}/common/math_util.c
		src/suite.c
		src/math_util.c
		src/vector.c
		src/mask.c
)

# Fixed point specific tests
if(NOT CONFIG_FPU)
	list(APPEND SOURCES src/fixed_point_int_sqrtf.c)
endif()

# Add the Zephyr unittest component
find_package(Zephyr COMPONENTS unittest REQUIRED HINTS $ENV{ZEPHYR_BASE})

# Link the stub/common headers to testbinary (generated by unittest)
cros_ec_stub_common(testbinary)
